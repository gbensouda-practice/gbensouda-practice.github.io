# ─── COUCHE 1 : LES OPÉRATIONS ─────────────────────────────────────
#  Ce que tu fais. Cette liste est faite pour NE PAS grandir : de
#  nouveaux projets apportent de nouvelles formes et de nouveaux
#  outils, pas de nouvelles opérations. Si tu ressens le besoin d'en
#  ajouter une, c'est le signe qu'une des cinq est mal définie.

operations :
  concevoir      # définir le principe et la forme
  systematiser   # construire une définition, un outil réutilisable
  fabriquer      # produire soi-même la pièce, matérielle ou numérique
  transmettre    # produire ce qui permet à d'autres de faire, et
                 # accompagner l'exécution. Ne compte pas la simple
                 # sous-traitance d'une étape : le livrable doit ÊTRE
                 # l'instruction.
  experimenter   # chercher sans commande, prototyper, tester


# ─── COUCHE 2 : LES FORMES ─────────────────────────────────────────
